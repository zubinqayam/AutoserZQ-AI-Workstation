smoke_test = '''#!/usr/bin/env node
/**
 * ZQ WORKSTATION v3.2.1 — PRODUCTION SMOKE TESTS
 * Run: node smoke_test.js
 * All tests must pass for production promotion.
 */

const http = require('http');

const BASE = process.env.ZQ_BASE_URL || 'http://localhost:3000';
let passed = 0, failed = 0;

async function test(name, fn) {
  try {
    await fn();
    console.log(`  ✅ PASS: ${name}`);
    passed++;
  } catch (e) {
    console.log(`  ❌ FAIL: ${name} — ${e.message}`);
    failed++;
  }
}

async function request(method, path, body) {
  return new Promise((resolve, reject) => {
    const url = new URL(path, BASE);
    const opts = { method, hostname: url.hostname, port: url.port, path: url.pathname + url.search, headers: {} };
    if (body) { opts.headers['Content-Type'] = 'application/json'; }
    const req = http.request(opts, (res) => {
      let data = '';
      res.on('data', chunk => data += chunk);
      res.on('end', () => {
        try { resolve({ status: res.statusCode, body: JSON.parse(data) }); }
        catch (e) { resolve({ status: res.statusCode, body: data }); }
      });
    });
    req.on('error', reject);
    if (body) req.write(JSON.stringify(body));
    req.end();
  });
}

async function runTests() {
  console.log('╔════════════════════════════════════════════════════════════════╗');
  console.log('║  ZQ WORKSTATION v3.2.1 — PRODUCTION SMOKE TESTS                ║');
  console.log('║  Target:', BASE.padEnd(48), '║');
  console.log('╚════════════════════════════════════════════════════════════════╝');

  // GATE 1: Infrastructure
  await test('Server startup — /api/health returns 200', async () => {
    const res = await request('GET', '/api/health');
    if (res.status !== 200) throw new Error(`Status ${res.status}`);
    if (!res.body.version) throw new Error('Missing version');
  });

  // GATE 2: Security
  await test('Input validation — rejects control characters', async () => {
    const res = await request('POST', '/api/rer', { topic: 'test\\x00injection' });
    // Should sanitize and proceed, or reject
    if (res.status !== 200 && res.status !== 400) throw new Error(`Unexpected ${res.status}`);
  });

  // GATE 3: Runtime — Frame Validation
  await test('Frame validation — google.com blocked', async () => {
    const res = await request('GET', '/api/validate-frame?url=https://google.com');
    if (res.status !== 200) throw new Error(`Status ${res.status}`);
    if (res.body.frameable !== false) throw new Error('Expected frameable:false');
  });

  await test('Frame validation — bing.com allowed', async () => {
    const res = await request('GET', '/api/validate-frame?url=https://bing.com');
    if (res.status !== 200) throw new Error(`Status ${res.status}`);
    if (res.body.frameable !== true) throw new Error('Expected frameable:true');
  });

  // GATE 3: Runtime — RER Execution
  let taskId;
  await test('RER execution — queues research task', async () => {
    const res = await request('POST', '/api/rer', { topic: 'AI Service Resilience' });
    if (res.status !== 200) throw new Error(`Status ${res.status}`);
    if (!res.body.taskId) throw new Error('Missing taskId');
    taskId = res.body.taskId;
  });

  await test('RER status — returns task with checkpoints', async () => {
    const res = await request('GET', `/api/rer/${taskId}`);
    if (res.status !== 200) throw new Error(`Status ${res.status}`);
    if (!res.body.checkpoints) throw new Error('Missing checkpoints');
  });

  // GATE 3: Runtime — Budget
  await test('Budget endpoint — returns usage stats', async () => {
    const res = await request('GET', '/api/budget');
    if (res.status !== 200) throw new Error(`Status ${res.status}`);
    if (typeof res.body.hourlyUsed !== 'number') throw new Error('Missing hourlyUsed');
  });

  // GATE 3: Runtime — Search Fallback
  await test('Search fallback — returns Bing URL', async () => {
    const res = await request('GET', '/api/search?q=artificial+intelligence');
    if (res.status !== 200) throw new Error(`Status ${res.status}`);
    if (!res.body.url.includes('bing.com')) throw new Error('Expected Bing URL');
  });

  // GATE 4: Recovery — DLQ
  await test('DLQ endpoint — returns array', async () => {
    const res = await request('GET', '/api/dlq');
    if (res.status !== 200) throw new Error(`Status ${res.status}`);
    if (!Array.isArray(res.body)) throw new Error('Expected array');
  });

  // Summary
  console.log('');
  console.log('╔════════════════════════════════════════════════════════════════╗');
  console.log(`║  RESULTS: ${passed} passed, ${failed} failed${''.padEnd(32 - String(passed).length - String(failed).length)}║`);
  console.log(failed === 0 ? '║  STATUS:  PRODUCTION READY ✅' : '║  STATUS:  BLOCKED — fix failures before promotion ❌');
  console.log('╚════════════════════════════════════════════════════════════════╝');
  process.exit(failed === 0 ? 0 : 1);
}

runTests().catch(e => { console.error('Smoke test runner error:', e); process.exit(1); });
'''
